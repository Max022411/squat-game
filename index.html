Unity C# 體感與邏輯控制腳本 (`SquatDetector.cs`) 若規劃移植至 **Unity 引擎** 並搭配 MediaPipe 插件，請使用以下具備平滑濾波與遲滯區間的 C# 控制腳本[cite: 1, 3]： ```csharp using UnityEngine; using UnityEngine.Events; public class SquatDetector : MonoBehaviour { private enum SquatState { Standing, Squatting } private SquatState currentState = SquatState.Standing; [Header("深蹲判定門檻設定 (遲滯區間)")] [Tooltip("髖關節低於膝蓋多少 Y 軸距離時判定為蹲下")] public float squatEnterThreshold = 0.05f; [Tooltip("髖關節高於膝蓋多少 Y 軸距離時判定為站起 (需大於 enterThreshold)")] public float squatExitThreshold = 0.20f; [Header("平滑濾波與死區防抖")] [Range(0.01f, 1f)] public float smoothingAlpha = 0.25f; // 指數平滑係數 public float minAttackInterval = 0.8f; // 攻擊冷卻時間 (秒) [Header("戰鬥與 Combo 事件")] public UnityEvent onSquatAttack; private Vector3 smoothedHip; private Vector3 smoothedKnee; private bool isInitialized = false; private float lastAttackTime; // 模擬或實時接收來自 MediaPipe 插件的關節世界座標 public void UpdateJointPositions(Vector3 rawHipPosition, Vector3 rawKneePosition) { if (!isInitialized) { smoothedHip = rawHipPosition; smoothedKnee = rawKneePosition; isInitialized = true; return; } // 1. 套用 EMA 指數平滑濾波，過濾鏡頭與 depth 估算帶來的抖動 smoothedHip = Vector3.Lerp(smoothedHip, rawHipPosition, smoothingAlpha); smoothedKnee = Vector3.Lerp(smoothedKnee, rawKneePosition, smoothingAlpha); // 2. 計算相對高度差 (Y 軸) float heightDifference = smoothedHip.y - smoothedKnee.y; // 3. 狀態機判定 EvaluateSquat(heightDifference); } private void EvaluateSquat(float heightDifference) { switch (currentState) { case SquatState.Standing: if (heightDifference < squatEnterThreshold) { currentState = SquatState.Squatting; } break; case SquatState.Squatting: if (heightDifference > squatExitThreshold) { currentState = SquatState.Standing; if (Time.time - lastAttackTime > minAttackInterval) { lastAttackTime = Time.time; onSquatAttack?.Invoke(); // 觸發攻擊事件 } } break; } } } ```[cite: 1, 3]
